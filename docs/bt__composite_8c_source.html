1/**
2 * @file bt_composite.c
3 * @brief Tick implementations for Sequence, Selector, and Parallel nodes.
4 *
5 * @ingroup BT_INTERNAL_TICKS
6 *
7 * @details
8 * ## Sequence and Selector — stateful/memory policy
9 *
10 * Both use `bt_node_t::running_child` to remember which child was RUNNING
11 * on the previous tick. On resume, iteration starts from that index rather
12 * than from child 0. This means:
13 *
14 * - Completed actions (SUCCESS) before the running child are not re-executed.
15 * - If the parent forces a restart (halt then re-tick), `running_child` is
16 * reset to 0 by @ref bt_node_halt.
17 *
18 * This policy is preferable for most embedded use-cases where action nodes
19 * carry ongoing work that must not be interrupted unnecessarily.
20 *
21 * ## Parallel — fully reactive policy
22 *
23 * Every child is ticked on every tree tick. Children that returned a
24 * terminal status in a previous tick have `initialized = false`, so
25 * @ref bt_node_tick will call their `on_init_fn` again and re-execute them.
26 * This makes conditions re-evaluate on every tick and is the standard way
27 * to implement "monitoring" trees:
28 *
29 * @code
30 * Parallel [threshold = N]
31 * ├── Condition A (re-evaluated every tick — reactive guard)
32 * └── Action B (continues across ticks while RUNNING)
33 * @endcode
34 *
35 * If Condition A fails on any tick, the Parallel immediately returns
36 * FAILURE and halts Action B.
37 */

44/**
45 * @brief Tick implementation for Sequence nodes.
46 *
47 * Iterates children starting from `node->running_child`:
48 * - BT_RUNNING → record index, return BT_RUNNING.
49 * - BT_FAILURE → halt any earlier initialised siblings, reset index,
50 * return BT_FAILURE.
51 * - BT_SUCCESS → advance to the next child.
52 *
53 * All children succeeded → reset index, return BT_SUCCESS.
54 *
55 * @param[in,out] node Sequence node being ticked.
56 * @param[in] ctx User context pointer.
57 * @return BT_SUCCESS, BT_FAILURE, BT_RUNNING, or BT_ERROR.
58 */
59bt_status_t bt_sequence_tick(bt_node_t *node, void *ctx)
60{
61 for (uint16_t i = node->running_child; i < node->child_count; i++) {
62 bt_status_t s = bt_node_tick(node->children[i], ctx);
63
64 switch (s) {
65 case BT_RUNNING:
66 node->running_child = i;
67 return BT_RUNNING;
68
69 case BT_FAILURE:
70 /* Halt any initialised siblings that preceded the failing child */
71 for (uint16_t j = 0; j < i; j++)
72 bt_node_halt(node->children[j], ctx);
73 node->running_child = 0;
74 return BT_FAILURE;
75
76 case BT_SUCCESS:
77 break; /* advance to next child */
78
79 default:
80 node->running_child = 0;
81 return BT_ERROR;
82 }
83 }
84
85 node->running_child = 0;
86 return BT_SUCCESS;
87}

93/**
94 * @brief Tick implementation for Selector nodes.
95 *
96 * Iterates children starting from `node->running_child`:
97 * - BT_RUNNING → record index, return BT_RUNNING.
98 * - BT_SUCCESS → halt earlier siblings, reset index, return BT_SUCCESS.
99 * - BT_FAILURE → try the next child.
100 *
101 * All children failed → reset index, return BT_FAILURE.
102 *
103 * @param[in,out] node Selector node being ticked.
104 * @param[in] ctx User context pointer.
105 * @return BT_SUCCESS, BT_FAILURE, BT_RUNNING, or BT_ERROR.
106 */
107bt_status_t bt_selector_tick(bt_node_t *node, void *ctx)
108{
109 for (uint16_t i = node->running_child; i < node->child_count; i++) {
110 bt_status_t s = bt_node_tick(node->children[i], ctx);
111
112 switch (s) {
113 case BT_RUNNING:
114 node->running_child = i;
115 return BT_RUNNING;
116
117 case BT_SUCCESS:
118 for (uint16_t j = 0; j < i; j++)
119 bt_node_halt(node->children[j], ctx);
120 node->running_child = 0;

122

125

140/**
141 * @brief Tick implementation for Parallel nodes.
142 *
143 * Every child is ticked unconditionally on each call (fully reactive).
144 * Children that completed (SUCCESS or FAILURE) in a previous tick have
145 * `initialized = false`; @ref bt_node_tick will re-initialise and re-run
146 * them, making conditions re-evaluate every tick.
147 *
148 * Decision logic:
149 * - success_count >= success_threshold → halt all children, return SUCCESS.
150 * - (child_count − failure_count) < success_threshold → success impossible,
151 * halt all children, return FAILURE.
152 * - Otherwise → return RUNNING.
153 *
154 * @param[in,out] node Parallel node being ticked.
155 * @param[in] ctx User context pointer.
156 * @return BT_SUCCESS, BT_FAILURE, or BT_RUNNING.
157 */
158bt_status_t bt_parallel_tick(bt_node_t *node, void *ctx)
159{
160 uint16_t success_count = 0;
161 uint16_t failure_count = 0;
162 const uint16_t n = node->child_count;
163 const uint16_t thresh = node->success_threshold;
164
165 for (uint16_t i = 0; i < n; i++) {
166 bt_status_t s = bt_node_tick(node->children[i], ctx);
167 if (s == BT_SUCCESS) success_count++;
168 else if (s == BT_FAILURE) failure_count++;
169 }
170
171 if (success_count >= thresh) {
172 for (uint16_t i = 0; i < n; i++)
173 bt_node_halt(node->children[i], ctx);
174 return BT_SUCCESS;
175 }
176
177 /* Success is now impossible — fail fast */
178 if ((uint16_t)(n - failure_count) < thresh) {
179 for (uint16_t i = 0; i < n; i++)
180 bt_node_halt(node->children[i], ctx);
181 return BT_FAILURE;
182 }
183
184 return BT_RUNNING;
185}
