3 * @brief Minimal FreeRTOS API simulation using POSIX pthreads.
4 *
5 * Allows the FreeRTOS example to compile and run on Linux / macOS without a
6 * real FreeRTOS environment. On a real embedded target, remove this file

56static inline SemaphoreHandle_t xSemaphoreCreateRecursiveMutex(void)
57{
58 pthread_mutexattr_t attr;
59 pthread_mutex_t *m = (pthread_mutex_t *)malloc(sizeof(pthread_mutex_t));
60 if (!m) return NULL;
61 pthread_mutexattr_init(&attr);
62 pthread_mutexattr_settype(&attr, PTHREAD_MUTEX_RECURSIVE_NP); /* Linux NP variant; no feature-test required */
63 if (pthread_mutex_init(m, &attr) != 0) { free(m); m = NULL; }
64 pthread_mutexattr_destroy(&attr);
65 return m;
66}

101static inline BaseType_t xTaskCreate(TaskFunction_t fn,
102 const char *name,
103 uint16_t stack_depth,
104 void *param,
105 UBaseType_t priority,
106 TaskHandle_t *out_handle)
107{
108 (void)name; (void)stack_depth; (void)priority;
109 _sim_task_arg_t *ta = (_sim_task_arg_t *)malloc(sizeof(_sim_task_arg_t));
110 if (!ta) return pdFAIL;
111 ta->fn = fn;
112 ta->arg = param;
113 pthread_t *t = (pthread_t *)malloc(sizeof(pthread_t));
114 if (!t) { free(ta); return pdFAIL; }
115 if (pthread_create(t, NULL, _sim_task_trampoline, ta) != 0) {
116 free(ta); free(t); return pdFAIL;
117 }
118 if (out_handle) *out_handle = (TaskHandle_t)t;
119 else pthread_detach(*t);
120 return pdPASS;
121}
