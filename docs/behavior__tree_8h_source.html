1/**
2 * @file behavior_tree.h
3 * @brief lib-behavior-tree — public API.
4 * See README.md (Doxygen mainpage) for the full architecture overview.
5 *
6 * @note Only this header needs to be included by application code.
7 *
8 * **lib-behavior-tree** is a generic, thread-safe, dynamically-allocated
9 * Behaviour Tree library written in pure C99. It targets embedded systems
10 * running bare-metal, FreeRTOS, or any POSIX-compatible RTOS.
11 *
12 * @section arch_sec Architecture
13 *
14 * @subsection tree_sec Tree structure
15 * A tree is a directed acyclic graph rooted at a single `bt_node_t`.
16 * Composite and decorator nodes own a dynamic array of child pointers.
17 * Leaf nodes (Action, Condition) have no children but carry user callbacks.
18 *
19 * @code
20 * bt_tree_t
21 * └── root : bt_node_t *
22 * ├── type (BT_NODE_SEQUENCE, BT_NODE_SELECTOR, …)
23 * ├── status (BT_SUCCESS / BT_FAILURE / BT_RUNNING / BT_INVALID)
24 * ├── children[] (dynamic array, grows on demand)
25 * ├── tick_fn (composite/decorator built-in, or user leaf fn)
26 * ├── on_init_fn (called once at start of each execution cycle)
27 * ├── on_halt_fn (called when node is forcibly interrupted)
28 * └── user_data (opaque payload forwarded to every callback)
29 * @endcode
30 *
31 * @subsection lifecycle_sec Node lifecycle
32 * @code
33 * NOT_STARTED ──bt_node_tick()──> on_init() called → tick_fn()
34 * │
35 * ┌───────────────────────────┼──────────────────────┐
36 * RUNNING SUCCESS FAILURE
37 * (initialized=true) (initialized=false) (initialized=false)
38 * │ │ │
39 * next tick resumes next tick calls next tick calls
40 * without on_init on_init again on_init again
41 * │
42 * bt_node_halt() ──> on_halt() called, initialized=false, status=INVALID
43 * @endcode
44 *
45 * @subsection composite_sec Composite node policies
46 * - **Sequence** and **Selector**: stateful/memory policy — resumes from the
47 * last RUNNING child rather than restarting from child 0 each tick. This
48 * avoids re-executing completed actions.
49 * - **Parallel**: fully reactive — all children are ticked on every call.
50 * Use Parallel to build monitoring patterns: place conditions alongside a
51 * long-running action; if any condition fails, the Parallel fails and the
52 * action is halted immediately.
53 *
54 * @subsection thread_sec Thread safety
55 * Each tree optionally owns a recursive mutex (`bt_port.h` supplies the
56 * platform-specific type and operations). When `thread_safe = true` is
57 * passed to `bt_tree_create()`, `bt_tree_tick()` holds the mutex for the
58 * entire duration of the tick. Node creation, `bt_node_add_child()`, and
59 * tree construction are NOT protected — build the tree before sharing it.
60 *
61 * @subsection memory_sec Memory
62 * All allocations go through `BT_MALLOC` / `BT_FREE`, overridable in
63 * `conf_behaviour_tree.h`. A `bt_node_t` is ~60–72 bytes.
64 *
65 * @section usage_sec Quick-start
66 * 1. Copy `inc/conf_behaviour_tree.h` to your project's include path.
67 * 2. Set `BT_PLATFORM` and optionally `BT_MALLOC` / `BT_FREE`.
68 * 3. Add `inc/` to your compiler's include search path.
69 * 4. Compile `src/behavior_tree.c`, `src/bt_composite.c`,
70 * `src/bt_decorator.c`, and exactly one `src/bt_port_<platform>.c`.
71 * 5. `#include "behavior_tree.h"` and use the factory functions.
72 *
73 * @author lib-behavior-tree contributors
74 * @version 1.0.0
75 */

93/**
94 * @brief Execution status returned by every node tick.
95 *
96 * Leaf callbacks must return one of BT_SUCCESS, BT_FAILURE, or BT_RUNNING.
97 * BT_INVALID and BT_ERROR are set by the library, never by user code.
98 */
99typedef enum {
100 BT_SUCCESS = 0, /**< Node completed its task successfully */
101 BT_FAILURE = 1, /**< Node completed but did not achieve its goal */
102 BT_RUNNING = 2, /**< Node is still executing (long-running operation) */
103 BT_INVALID = 3, /**< Node has not been ticked / was halted */
104 BT_ERROR = 4, /**< Internal error (NULL pointer, alloc failure, etc.) */
105} bt_status_t;
106
107/**
108 * @brief Structural role of a node within the tree.
109 */
110typedef enum {
111 /* --- Leaf nodes ---------------------------------------------------- */
112 BT_NODE_ACTION = 0, /**< User-defined action (may return RUNNING) */
113 BT_NODE_CONDITION = 1, /**< Instant check; returns SUCCESS or FAILURE */
114 /* --- Composite nodes ----------------------------------------------- */
115 BT_NODE_SEQUENCE = 2, /**< AND – children executed left-to-right */
116 BT_NODE_SELECTOR = 3, /**< OR – tries children until one succeeds */
117 BT_NODE_PARALLEL = 4, /**< ALL – all children ticked every tick */
118 /* --- Decorator nodes ----------------------------------------------- */
119 BT_NODE_INVERTER = 5, /**< Flips SUCCESS ↔ FAILURE */
120 BT_NODE_REPEATER = 6, /**< Loops child N times (or forever) */
121 BT_NODE_RETRY = 7, /**< Retries child on FAILURE up to N times */
122 BT_NODE_FORCE_SUCCESS = 8, /**< Maps any terminal status to SUCCESS */
123 BT_NODE_FORCE_FAILURE = 9, /**< Maps any terminal status to FAILURE */
124} bt_node_type_t;

135/**
136 * @brief Main tick callback — the node's per-cycle execution logic.
137 *
138 * Must return BT_SUCCESS, BT_FAILURE, or BT_RUNNING. Must not block.
139 * Use BT_RUNNING to signal work in progress; the tree scheduler will call
140 * this function again on the next @ref bt_tree_tick invocation.
141 *
142 * @param[in] node The node being ticked.
143 * @param[in] context User context pointer forwarded from @ref bt_tree_t.
144 * @return BT_SUCCESS, BT_FAILURE, or BT_RUNNING.
145 */
146typedef bt_status_t (*bt_tick_fn_t) (struct bt_node *node, void *context);
147
148/**
149 * @brief Called once at the beginning of each new execution cycle.
150 *
151 * Invoked by @ref bt_node_tick the first time a node is ticked after it
152 * was created or after a previous cycle completed (returned SUCCESS/FAILURE)
153 * or the node was halted. Use it to reset state-machine variables, timers,
154 * counters, or any other per-cycle state.
155 *
156 * @param[in] node The node being initialised.
157 * @param[in] context User context pointer.
158 */
159typedef void (*bt_on_init_fn_t)(struct bt_node *node, void *context);
160
161/**
162 * @brief Called when the node is forcibly interrupted before completing.
163 *
164 * Invoked by @ref bt_node_halt when a parent composite decides to abort a
165 * RUNNING child (e.g., a Sequence sibling failed, or the tree was destroyed).
166 * Use it to cancel ongoing hardware operations, release locks, or free
167 * temporary resources.
168 *
169 * @param[in] node The node being halted.
170 * @param[in] context User context pointer.
171 */
172typedef void (*bt_on_halt_fn_t)(struct bt_node *node, void *context);

181/**
182 * @brief Single node in the behaviour tree.
183 *
184 * Composite and decorator nodes own a heap-allocated array of child pointers
185 * that grows automatically. Leaf nodes have @c child_count == 0.
186 *
187 * @note Do not modify fields directly; use the factory and manipulation
188 * functions instead. Internal fields (initialized, running_child,
189 * repeat_count) are managed by the library.
190 */
191typedef struct bt_node {
192 bt_node_type_t type; /**< Structural role of this node */
193 bt_status_t status; /**< Status after the most recent tick */
194
195 /* --- Tree hierarchy ------------------------------------------------- */
196 struct bt_node *parent; /**< Parent node (NULL for root) */
197 struct bt_node **children; /**< Heap-allocated child pointer array */
198 uint16_t child_count; /**< Number of current children */
199 uint16_t child_capacity; /**< Allocated capacity of children[] */
200
201 /* --- User callbacks ------------------------------------------------- */
202 bt_tick_fn_t tick_fn; /**< Mandatory execution callback */
203 bt_on_init_fn_t on_init_fn; /**< Optional cycle-start callback */
204 bt_on_halt_fn_t on_halt_fn; /**< Optional interrupt callback */
205
206 /* --- User payload --------------------------------------------------- */
207 void *user_data; /**< Forwarded to every callback */
208
209 /* --- Internal execution state --------------------------------------- */
210 bool initialized; /**< TRUE while mid-cycle (on_init ran) */
211 uint16_t running_child; /**< Stateful composite: resume index */
212
213 /* --- Decorator / Parallel parameters -------------------------------- */
214 int32_t repeat_count; /**< Current repeat / retry iteration */
215 int32_t max_count; /**< Maximum iterations (-1 = infinite) */
216 uint16_t success_threshold; /**< Parallel: min successes to pass */
217
218#if BT_ENABLE_NODE_NAMES
219 const char *name; /**< Human-readable label (debug only) */
220#endif
221} bt_node_t;

274/**
275 * @brief Halt and free the tree and every node it owns.
276 *
277 * Calls @ref bt_node_halt on the root (propagates to all RUNNING descendants),
278 * then recursively frees all nodes, then frees the tree itself. Destroys the
279 * mutex if one was created.
280 *
281 * @param[in] tree Tree to destroy. May be @c NULL (no-op).
282 *
283 * @warning Do not call @ref bt_tree_tick after this.
284 */
285void bt_tree_destroy(bt_tree_t *tree);

287/**
288 * @brief Execute one tick of the behaviour tree.
289 *
290 * Acquires the tree mutex (if thread_safe), ticks the root node, updates
291 * statistics if enabled, releases the mutex, and returns the root status.
292 *
293 * Call this at a fixed rate from your control loop or an RTOS task:
294 * @code
295 * while (1) {
296 * vTaskDelay(pdMS_TO_TICKS(20));
297 * bt_tree_tick(tree);
298 * }
299 * @endcode
300 *
301 * @param[in] tree Tree to tick.
302 * @return Root node status: BT_SUCCESS, BT_FAILURE, BT_RUNNING,
303 * or BT_ERROR if @p tree / root is NULL.
304 */
305bt_status_t bt_tree_tick (bt_tree_t *tree);

315/**
316 * @brief Create a leaf **Action** node.
317 *
318 * Actions represent ongoing operations. Their @p tick_fn may return
319 * BT_RUNNING to signal work in progress; on the next tree tick @p tick_fn
320 * is called again (without calling @p on_init_fn again) until the action
321 * returns a terminal status.
322 *
323 * @param[in] name Human-readable label (stored by pointer; use a
324 * string literal). Ignored if BT_ENABLE_NODE_NAMES=0.
325 * @param[in] tick_fn Mandatory execution callback (must not be NULL).
326 * @param[in] on_init_fn Called at the start of each execution cycle.
327 * Pass NULL if not needed.
328 * @param[in] on_halt_fn Called when the node is forcibly interrupted.
329 * Pass NULL if not needed.
330 * @param[in] user_data Available through node->user_data in callbacks.
331 * @return Pointer to new node, or NULL on allocation failure.
332 */
333bt_node_t *bt_action_create (const char *name,
334 bt_tick_fn_t tick_fn,
335 bt_on_init_fn_t on_init_fn,
336 bt_on_halt_fn_t on_halt_fn,
337 void *user_data);

339/**
340 * @brief Create a leaf **Condition** node.
341 *
342 * Conditions are instant checks; @p tick_fn must return either BT_SUCCESS
343 * or BT_FAILURE, never BT_RUNNING. No @p on_halt_fn is registered because
344 * conditions cannot be in the RUNNING state.
345 *
346 * @param[in] name Human-readable label.
347 * @param[in] tick_fn Condition check callback (must not be NULL).
348 * @param[in] user_data Forwarded to callbacks via node->user_data.
349 * @return Pointer to new node, or NULL on allocation failure.
350 */
351bt_node_t *bt_condition_create(const char *name,
352 bt_tick_fn_t tick_fn,
353 void *user_data);

355/**
356 * @brief Create a **Sequence** (→) composite node.
357 *
358 * Ticks children left-to-right (stateful/memory policy):
359 * - Returns BT_SUCCESS if **all** children return BT_SUCCESS.
360 * - Returns BT_FAILURE on the first child that returns BT_FAILURE.
361 * - Returns BT_RUNNING while a child is RUNNING; resumes from that child
362 * on the next tick without re-executing earlier children.
363 *
364 * @param[in] name Human-readable label.
365 * @return Pointer to new node, or NULL on allocation failure.
366 */
367bt_node_t *bt_sequence_create (const char *name);
368
369/**
370 * @brief Create a **Selector** (?) composite node.
371 *
372 * Ticks children left-to-right (stateful/memory policy):
373 * - Returns BT_SUCCESS on the first child that returns BT_SUCCESS.
374 * - Returns BT_FAILURE if **all** children return BT_FAILURE.
375 * - Returns BT_RUNNING while a child is RUNNING; resumes from that child
376 * on the next tick.
377 *
378 * @param[in] name Human-readable label.
379 * @return Pointer to new node, or NULL on allocation failure.
380 */
381bt_node_t *bt_selector_create (const char *name);
382
383/**
384 * @brief Create a **Parallel** composite node.
385 *
386 * Ticks **all** children on every tree tick (fully reactive). This makes it
387 * ideal for safety monitoring: combine conditions with long-running actions;
388 * if any condition fails the whole parallel fails and the actions are halted.
389 *
390 * @code
391 * Parallel [threshold = N]
392 * ├── Condition A (re-evaluated every tick)
393 * ├── Condition B (re-evaluated every tick)
394 * └── Action C (continues its RUNNING state across ticks)
395 * @endcode
396 *
397 * @param[in] name Human-readable label.
398 * @param[in] success_threshold Number of children that must return
399 * BT_SUCCESS for the Parallel to return
400 * BT_SUCCESS. The node returns BT_FAILURE when
401 * @c (child_count - failures) < threshold,
402 * making success impossible.
403 * @return Pointer to new node, or NULL on allocation failure.
404 */
405bt_node_t *bt_parallel_create (const char *name, uint16_t success_threshold);
