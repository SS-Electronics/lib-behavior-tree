1/**
2 * @file bt_motor_controller.c
3 * @brief FreeRTOS example: 3-phase motor-drive safety supervisor.
4 *
5 * @details
6 * Demonstrates lib-behavior-tree running inside a FreeRTOS application with
7 * two concurrent tasks:
8 *
9 * - **vSensorTask** (10 ms period): Reads ADC values (current, temperature)
10 * and a digital input (emergency-stop) into a shared @ref motor_ctx_t.
11 * - **vBtTask** (20 ms period): Ticks the behaviour tree. The tree mutex
12 * provides the thread-safety boundary between the tasks.
13 *
14 * Behaviour tree topology
15 * @code
16 * Selector [root]
17 * ├── Parallel [monitored_op, threshold=3] ← all 3 branches must not fail
18 * │ ├── Condition : estop_clear ← re-evaluated every tick
19 * │ ├── Condition : temp_ok ← re-evaluated every tick
20 * │ └── Sequence [run_motor] ← stateful (memory) action chain
21 * │ ├── Action : enable_drive (one-shot SUCCESS; idempotent)
22 * │ └── Action : ramp_speed (RUNNING while ramping, then holds RUNNING)
23 * └── Sequence [fault_response]
24 * ├── Action : disable_drive
25 * └── Action : set_fault_led
26 * @endcode
27 *
28 * The Parallel node re-ticks all three children on every tree tick.
29 * As soon as either condition returns FAILURE (E-stop pressed, over-temp),
30 * the Parallel fails, which causes the Selector to fall through to the
31 * fault-response branch. This is the canonical BT safety-monitoring pattern.
32 *
33 * Simulation
34 * ----------
35 * The example ships with freertos_sim.h, a thin POSIX wrapper that maps
36 * FreeRTOS types and APIs to pthreads so the code can be compiled and run on
37 * Linux / macOS without real hardware.
38 *
39 * @code
40 * gcc -std=c99 -Wall -Wextra \
41 * -I../../inc -I. \
42 * ../../src/behavior_tree.c \
43 * ../../src/bt_composite.c \
44 * ../../src/bt_decorator.c \
45 * ../../src/bt_port_freertos.c \
46 * bt_motor_controller.c \
47 * -lpthread -o motor_controller
48 * ./motor_controller
49 * @endcode
50 *
51 * Porting to real hardware
52 * ------------------------
53 * 1. Remove freertos_sim.h / FreeRTOS.h / semphr.h from the include path.
54 * 2. Replace the simulated sensor reads with actual HAL / driver calls.
55 * 3. Replace the simulated actuator writes with PWM / GPIO writes.
56 * 4. Adjust task stack sizes and priorities for your RTOS configuration.
57 */
58

165/**
166 * @brief Action — linearly ramp motor speed to SPEED_TARGET_RPM.
167 *
168 * Returns BT_RUNNING while still accelerating.
169 * Returns BT_SUCCESS when the setpoint is reached.
170 */
171static bt_status_t action_ramp_speed(bt_node_t *node, void *ctx)
172{
173 (void)node;
174 motor_ctx_t *m = (motor_ctx_t *)ctx;
175
176 if (m->speed_actual < SPEED_TARGET_RPM) {
177 m->speed_actual += RAMP_RATE_RPM_PER_TICK;
178 if (m->speed_actual > SPEED_TARGET_RPM)
179 m->speed_actual = SPEED_TARGET_RPM;
180 printf(" [ACT ] Ramping: %.0f / %.0f RPM\n",
181 m->speed_actual, SPEED_TARGET_RPM);
182 return BT_RUNNING;
183 }
184
185 /* Hold at setpoint — return RUNNING to keep the Parallel alive. */
186 return BT_RUNNING;
187}

237/**
238 * @brief Build the motor-controller behaviour tree.
239 *
240 * @code
241 * Selector [root]
242 * ├── Sequence [normal_op]
243 * │ ├── Condition : estop_clear
244 * │ ├── Condition : temp_ok
245 * │ ├── Action : enable_drive
246 * │ └── Action : ramp_speed
247 * └── Sequence [fault_response]
248 * ├── Action : disable_drive
249 * └── Action : set_fault_led
250 * @endcode
251 *
252 * thread_safe = true so bt_tree_tick can be called from vBtTask while
253 * vSensorTask simultaneously updates the shared context (after acquiring
254 * the same ctx_mutex).
255 */
256static bt_tree_t *build_motor_tree(motor_ctx_t *ctx)
257{
258 /* --- Motor run sequence (stateful/memory: enable once, ramp, hold) --- */
259 bt_node_t *act_en = bt_action_create("enable_drive",
260 action_enable_drive,
261 NULL, NULL, NULL);
262 bt_node_t *act_ramp = bt_action_create("ramp_speed",
263 action_ramp_speed,
264 action_ramp_init,
265 action_ramp_halt,
266 NULL);
267 bt_node_t *seq_run = bt_sequence_create("run_motor");
268 bt_node_add_child(seq_run, act_en);
269 bt_node_add_child(seq_run, act_ramp);
270
271 /* --- Parallel monitor: all three branches must not fail.
272 * threshold = 3 → fails if ANY of the three branches fails.
273 * The two conditions are re-evaluated on every tick (reactive).
274 * The motor sequence runs alongside and is halted on any fault. --- */
275 bt_node_t *cond_es = bt_condition_create("estop_clear",
276 cond_estop_clear, NULL);
277 bt_node_t *cond_temp = bt_condition_create("temp_ok",
278 cond_temp_ok, NULL);
279 bt_node_t *par_mon = bt_parallel_create("monitored_op", /*threshold=*/3);
280 bt_node_add_child(par_mon, cond_es);
281 bt_node_add_child(par_mon, cond_temp);
282 bt_node_add_child(par_mon, seq_run);
283
284 /* --- Fault response branch --- */
285 bt_node_t *act_dis = bt_action_create("disable_drive",
286 action_disable_drive,
287 NULL, NULL, NULL);
288 bt_node_t *act_led = bt_action_create("set_fault_led",
289 action_fault_led,
290 NULL, NULL, NULL);
291 bt_node_t *seq_fault = bt_sequence_create("fault_response");
292 bt_node_add_child(seq_fault, act_dis);
293 bt_node_add_child(seq_fault, act_led);
294
295 /* --- Root selector: try normal operation, fall back to fault response --- */
296 bt_node_t *sel_root = bt_selector_create("root");
297 bt_node_add_child(sel_root, par_mon);
298 bt_node_add_child(sel_root, seq_fault);
299
300 return bt_tree_create(sel_root, ctx, /*thread_safe=*/true);
301}

313/**
314 * @brief Sensor task — simulates ADC reads and digital input sampling.
315 *
316 * Runs at SENSOR_TASK_PERIOD_MS. Acquires ctx_mutex before writing to
317 * prevent torn reads by vBtTask.
318 *
319 * In this simulation:
320 * - Temperature rises slowly.
321 * - E-stop fires briefly at t = 600 ms.
322 * - Temperature over-limit fires at t = 900 ms.
323 */
324static void vSensorTask(void *param)
325{
326 task_params_t *p = (task_params_t *)param;
327 motor_ctx_t *ctx = p->ctx;
328 uint32_t tick = 0;
329
330 printf("[SensorTask] started\n");
331
332 while (tick * SENSOR_TASK_PERIOD_MS < SIM_DURATION_MS) {
333 vTaskDelay(pdMS_TO_TICKS(SENSOR_TASK_PERIOD_MS));
334 tick++;
335
336 xSemaphoreTakeRecursive(ctx->ctx_mutex, portMAX_DELAY);
337
338 /* Simulate rising temperature */
339 ctx->temperature_c += 0.5f;
340
341 /* Simulate a 200 ms E-stop event at 600 ms */
342 uint32_t t_ms = tick * SENSOR_TASK_PERIOD_MS;
343 ctx->estop_active = (t_ms >= 600 && t_ms < 800);
344
345 /* Simulate thermal shutdown after 900 ms */
346 if (t_ms >= 900) ctx->temperature_c = 90.0f;
347
348 /* Simulate phase current reading */
349 ctx->phase_current_a = ctx->drive_enabled ? 12.5f : 0.0f;
350
351 xSemaphoreGiveRecursive(ctx->ctx_mutex);
352 }
353
354 printf("[SensorTask] simulation complete\n");
355}

357/**
358 * @brief Behaviour-tree task — ticks the tree at BT_TASK_PERIOD_MS.
359 *
360 * The tree was created with thread_safe = true, so bt_tree_tick internally
361 * holds the tree mutex. Sensor data protected by ctx_mutex (different lock)
362 * is updated between ticks.
363 */
364static void vBtTask(void *param)
365{
366 task_params_t *p = (task_params_t *)param;
367 bt_tree_t *tree = p->tree;
368 motor_ctx_t *ctx = p->ctx;
369 uint32_t tick = 0;
370
371 printf("[BtTask] started (period: %u ms)\n", BT_TASK_PERIOD_MS);
372
373 while (tick * BT_TASK_PERIOD_MS < SIM_DURATION_MS) {
374 vTaskDelay(pdMS_TO_TICKS(BT_TASK_PERIOD_MS));
375 tick++;
376
377 /* Acquire ctx_mutex to snapshot consistent sensor data, then tick */
378 xSemaphoreTakeRecursive(ctx->ctx_mutex, portMAX_DELAY);
379 bt_status_t s = bt_tree_tick(tree);
380 xSemaphoreGiveRecursive(ctx->ctx_mutex);
381
382 printf("[BtTask] tick %3u → %-8s speed=%.0f RPM temp=%.1f°C "
383 "estop=%d drive=%d\n",
384 tick, bt_status_str(s),
385 ctx->speed_actual, ctx->temperature_c,
386 (int)ctx->estop_active, (int)ctx->drive_enabled);
387 }
388
389 printf("[BtTask] simulation complete (%u total ticks)\n", tick);
390#if BT_ENABLE_STATS
391 printf("[BtTask] tree tick count=%u last_status=%s\n",
392 tree->tick_count, bt_status_str(tree->last_status));
393#endif
394}

400int main(void)
401{
402 printf("=== Motor Controller — lib-behavior-tree FreeRTOS Example ===\n\n");
403
404 /* Initialise shared context */
405 motor_ctx_t ctx;
406 memset(&ctx, 0, sizeof(ctx));
407 ctx.temperature_c = 25.0f;
408 ctx.speed_setpoint = SPEED_TARGET_RPM;
409
410 ctx.ctx_mutex = xSemaphoreCreateRecursiveMutex();
411 if (!ctx.ctx_mutex) {
412 fprintf(stderr, "Failed to create context mutex\n");
413 return 1;
414 }
415
416 /* Build the tree (thread_safe=true creates its own internal mutex) */
417 bt_tree_t *tree = build_motor_tree(&ctx);
418 if (!tree) {
419 fprintf(stderr, "Failed to build behaviour tree\n");
420 vSemaphoreDelete(ctx.ctx_mutex);
421 return 1;
422 }
423
424 /* Task parameters (stack-allocated; both tasks finish before main exits) */
425 task_params_t params = { .tree = tree, .ctx = &ctx };
426
427 /* Create FreeRTOS tasks */
428 TaskHandle_t sensor_handle, bt_handle;
429
430 if (xTaskCreate(vSensorTask, "SensorTask", 512, &params, 2,
431 &sensor_handle) != pdPASS) {
432 fprintf(stderr, "Failed to create sensor task\n");
433 return 1;
434 }
435 if (xTaskCreate(vBtTask, "BtTask", 1024, &params, 3,
436 &bt_handle) != pdPASS) {
437 fprintf(stderr, "Failed to create BT task\n");
438 return 1;
439 }
440
441 /* In a real FreeRTOS application vTaskStartScheduler() never returns.
442 * In simulation we join the created pthreads to wait for completion. */
443 pthread_join(*((pthread_t *)sensor_handle), NULL);
444 pthread_join(*((pthread_t *)bt_handle), NULL);
445
446 /* Teardown */
447 bt_tree_destroy(tree);
448 vSemaphoreDelete(ctx.ctx_mutex);
449
450 printf("\n=== Simulation finished ===\n");
451 return 0;
452}
