1/**
2 * @file bt_port_posix.c
3 * @brief POSIX pthreads port — recursive mutex via `pthread_mutex_t`.
4 *
5 * @ingroup BT_PORT
6 *
7 * @details
8 * Compile this file when `BT_PLATFORM == BT_PLATFORM_POSIX`.
9 * Link your application with `-lpthread`.
10 *
11 * A **recursive** mutex attribute is set so that callbacks executing inside
12 * `bt_tree_tick` can safely call library utilities that also acquire the
13 * mutex (e.g., a custom composite calling `bt_node_tick` on a sub-tree).
14 *
15 * **Feature-test note:** `pthread_mutexattr_settype` and
16 * `PTHREAD_MUTEX_RECURSIVE` are XSI extensions and require either
17 * `-D_POSIX_C_SOURCE=200809L` or `-D_GNU_SOURCE` on the compiler command
18 * line when building with `-std=c99`. The example Makefile already includes
19 * this flag.
20 */

26/**
27 * @brief Initialise a recursive POSIX mutex.
28 *
29 * @param[out] mutex Pointer to the `pthread_mutex_t` to initialise.
30 * @return @c true on success; @c false if `pthread_mutex_init` fails.
31 */
32bool bt_port_mutex_create(bt_mutex_t *mutex)
33{
34 pthread_mutexattr_t attr;
35 if (pthread_mutexattr_init(&attr) != 0) return false;
36 pthread_mutexattr_settype(&attr, PTHREAD_MUTEX_RECURSIVE);
37 bool ok = (pthread_mutex_init(mutex, &attr) == 0);
38 pthread_mutexattr_destroy(&attr);
39 return ok;
40}
