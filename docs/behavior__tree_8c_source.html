30/**
31 * @brief Allocate and zero-initialise a bt_node_t with common fields set.
32 *
33 * @param[in] type Node type enum value.
34 * @param[in] name Human-readable label (stored by pointer).
35 * @param[in] tick_fn Tick callback to install.
36 * @return Pointer to initialised node, or NULL on allocation failure.
37 */
38static bt_node_t *node_alloc(bt_node_type_t type, const char *name,
39 bt_tick_fn_t tick_fn)
40{
41 bt_node_t *n = (bt_node_t *)BT_MALLOC(sizeof(bt_node_t));
42 if (!n) return NULL;
43 memset(n, 0, sizeof(bt_node_t));
44 n->type = type;
45 n->status = BT_INVALID;
46 n->max_count = -1;
47 n->tick_fn = tick_fn;
48#if BT_ENABLE_NODE_NAMES
49 n->name = name;
50#else
51 (void)name;
52#endif
53 return n;
54}

56/* ===========================================================================
57 * bt_node_tick
58 * ========================================================================= */
59
60/**
61 * @brief Tick a single node (see behavior_tree.h for full documentation).
62 *
63 * Execution cycle per node:
64 * 1. If @c node->initialized is @c false: call @c on_init_fn (if set),
65 * set @c initialized = @c true.
66 * 2. Call @c tick_fn and record the returned status.
67 * 3. If the status is terminal (SUCCESS or FAILURE), set
68 * @c initialized = @c false so the next call restarts the cycle.
69 */
70bt_status_t bt_node_tick(bt_node_t *node, void *context)
71{
72 if (!node || !node->tick_fn) return BT_ERROR;
73
74 if (!node->initialized) {
75 if (node->on_init_fn) node->on_init_fn(node, context);
76 node->initialized = true;
77 }
78
79 bt_status_t s = node->tick_fn(node, context);
80 node->status = s;
81
82 if (s != BT_RUNNING) node->initialized = false;
83
84 return s;
85}

87/* ===========================================================================
88 * bt_node_halt
89 * ========================================================================= */
90
91/**
92 * @brief Forcibly interrupt a node (see behavior_tree.h for full docs).
93 *
94 * Implementation notes:
95 * - The function is a no-op when @c node->initialized is @c false because
96 * a non-initialized node is not mid-cycle and needs no cleanup.
97 * - Recursion propagates halt to all initialized descendants first, then
98 * calls this node's @c on_halt_fn, matching the tear-down order that a
99 * user would expect (leaves first, then their parents).
100 */
101void bt_node_halt(bt_node_t *node, void *context)
102{
103 if (!node || !node->initialized) return;
104
105 for (uint16_t i = 0; i < node->child_count; i++)
106 bt_node_halt(node->children[i], context);
107
108 if (node->on_halt_fn) node->on_halt_fn(node, context);
109
110 node->status = BT_INVALID;
111 node->initialized = false;
112 node->running_child = 0;
113 node->repeat_count = 0;
114}

205/**
206 * @brief Append a child node (see behavior_tree.h for full documentation).
207 *
208 * Growth strategy: capacity doubles on each reallocation, starting from
209 * @c BT_CHILDREN_INITIAL_CAPACITY. The old array is freed after copying.
210 */
211bt_status_t bt_node_add_child(bt_node_t *parent, bt_node_t *child)
212{
213 if (!parent || !child) return BT_ERROR;
214
215 if (parent->child_count >= parent->child_capacity) {
216 uint16_t new_cap = (parent->child_capacity == 0)
217 ? (uint16_t)BT_CHILDREN_INITIAL_CAPACITY
218 : (uint16_t)(parent->child_capacity * 2u);
219
220 bt_node_t **arr = (bt_node_t **)BT_MALLOC(
221 (size_t)new_cap * sizeof(bt_node_t *));
222 if (!arr) return BT_ERROR;
223
224 if (parent->children) {
225 for (uint16_t i = 0; i < parent->child_count; i++)
226 arr[i] = parent->children[i];
227 BT_FREE(parent->children);
228 }
229 parent->children = arr;
230 parent->child_capacity = new_cap;
231 }
232
233 parent->children[parent->child_count++] = child;
234 child->parent = parent;
235 return BT_SUCCESS;
236}

242void bt_node_destroy(bt_node_t *node)
243{
244 if (!node) return;
245 for (uint16_t i = 0; i < node->child_count; i++)
246 bt_node_destroy(node->children[i]);
247 if (node->children) BT_FREE(node->children);
248 BT_FREE(node);

255bt_tree_t *bt_tree_create(bt_node_t *root, void *context, bool thread_safe)
256{
257 bt_tree_t *tree = (bt_tree_t *)BT_MALLOC(sizeof(bt_tree_t));
258 if (!tree) return NULL;
259 memset(tree, 0, sizeof(bt_tree_t));
260
261 tree->root = root;
262 tree->context = context;
263 tree->thread_safe = thread_safe;
264
265#if BT_ENABLE_STATS
266 tree->last_status = BT_INVALID;
267#endif
268
269 if (thread_safe && !bt_port_mutex_create(&tree->mutex)) {
270 BT_FREE(tree);
271 return NULL;
272 }
273
274 return tree;
275}

277void bt_tree_destroy(bt_tree_t *tree)
278{
279 if (!tree) return;
280
281 if (tree->thread_safe) bt_port_mutex_lock(&tree->mutex);
282
283 bt_node_halt (tree->root, tree->context);
284 bt_node_destroy(tree->root);
285
286 if (tree->thread_safe) {
287 bt_port_mutex_unlock (&tree->mutex);
288 bt_port_mutex_destroy(&tree->mutex);
289 }
290
291 BT_FREE(tree);
292}
293
294bt_status_t bt_tree_tick(bt_tree_t *tree)
295{
296 if (!tree || !tree->root) return BT_ERROR;
297
298 if (tree->thread_safe) bt_port_mutex_lock(&tree->mutex);
299
300 bt_status_t s = bt_node_tick(tree->root, tree->context);
301
302#if BT_ENABLE_STATS
303 tree->tick_count++;
304 tree->last_status = s;
305#endif
306
307 if (tree->thread_safe) bt_port_mutex_unlock(&tree->mutex);
308
309 return s;
310}
